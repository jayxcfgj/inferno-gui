# hier vorher den user die wichtigsten eckdaten abfragen (mit dem hinweis, dasser das auch später noch ändern kann)
# - welches sample rate er verwenden will (default 48000)
# - wie viele ein- und ausgabe-kanäle er haben möchte

# anschließend diese eingaben in der konfiugration einarbeiten

nano ~/.asoundrc

# Füge folgenden Inhalt ein:

pcm.inferno {
    type inferno

    NAME "Inferno"
    SAMPLE_RATE "48000"

    TX_CHANNELS 24
    RX_CHANNELS 24

    BIND_IP "<interface eintragen>"

    hint {
        show on
        description "Inferno Dante Virtual Soundcard"
    }
}

# PipeWire für Inferno freischalten

# Inferno benötigt bestimmte Systemaufrufe zur Uhrzeitsynchronisierung. Bei einem über systemd gestarteten PipeWire kann die # # Sicherheitsfilterung diese blockieren.

# Wir richten deshalb die vom Projekt vorgesehene Ausnahme ein.

mkdir -p ~/.config/systemd/user/pipewire.service.d

nano ~/.config/systemd/user/pipewire.service.d/override.conf

# Einfügen:

[Service]
SystemCallFilter=@clock


# Speichern und schließen.
# Die Datei entspricht der vom Inferno-Projekt bereitgestellten systemd-Konfiguration. 
# GitLab+1

systemctl --user restart pipewire
# Danach:
systemctl --user restart wireplumber
# Kontrolliere:
systemctl --user status pipewire
